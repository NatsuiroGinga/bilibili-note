## 已核实的事实（实现前提，勿重新推导）

- 严格过去接口四字段中三项可派生（见 spec 3.0 节）：`previous_segment_row` 由
  `(E23, T23)` 字典序排序得到，实测同实体相邻片段 `T23` **严格递增比例 100.000000%**；
  `source_record_id` 用 `X23` 行号（唯一稳定，但**非时间序**，前 20,000 片段中行内
  索引严格升序的仅 2,457 个，禁止用行号比先后）；`role_id` 由切分收据给出。
- 因果序两级：**片段内用列位置 0…127**，**片段间用 `T23`**。
- 作用面：99.0786% 的流有历史可读；多片段实体仅 2.22% 却覆盖 96.09% 的流，
  故跨片段 carry 必须实现，不能只做片段内。
- 复杂度：交叉注意力新增参数约 `150,913`（`R=8`），占裸 FT `924,283` 的 `16.33%`；
  每流前向 FLOP 增量仅 `0.70%`；状态显存 `R=8` 时约 `1,042.7 MB`（fp32，全实体）。

```python
def build_interface(cache_root: Path, split_receipt: Path) -> dict[str, np.ndarray]:
    """按 (实体, 片段时间戳) 建立同实体前驱片段索引。

    因果序两级：片段内由列位置给出，片段间由 T23 给出。X23 行号唯一但非时间序，
    不得用于比较先后。
    """
    entity = np.load(cache_root / "E23.npy")
    stamp = np.load(cache_root / "T23.npy")
    order = np.lexsort((stamp, entity))
    ordered_entity = entity[order]
    ordered_stamp = stamp[order]
    previous = np.full(len(entity), -1, dtype=np.int64)
    same = np.r_[False, ordered_entity[1:] == ordered_entity[:-1]]
    previous[order[same]] = order[np.flatnonzero(same) - 1]
    ordinal = np.zeros(len(entity), dtype=np.int64)
    starts = np.flatnonzero(~same)
    for begin, end in zip(starts, np.r_[starts[1:], len(order)]):
        ordinal[order[begin:end]] = np.arange(end - begin)
    return {
        "previous_segment_row": previous,
        "is_entity_start": previous < 0,
        "segment_ordinal": ordinal,
        "role_id": load_role_ids(split_receipt, len(entity)),
    }
```

```python
def assert_causality(interface: dict, entity: np.ndarray, stamp: np.ndarray) -> dict:
    """四条必须全过的断言；任一失败即抛错，不静默降级。"""
    previous = interface["previous_segment_row"]
    linked = previous >= 0
    checks = {
        "same_entity": bool(np.all(entity[previous[linked]] == entity[linked])),
        "strictly_increasing_time": bool(np.all(stamp[previous[linked]] < stamp[linked])),
        "same_role": bool(np.all(interface["role_id"][previous[linked]] == interface["role_id"][linked])),
        "no_self_reference": bool(np.all(previous[linked] != np.flatnonzero(linked))),
    }
    if not all(checks.values()):
        raise RuntimeError(f"因果接口断言失败：{checks}")
    return checks
```

- [ ] **步骤 1：实现记忆状态（槽 0 为 Welford 均值，槽 1..R-1 为最近队列）**

```python
class EntityMemoryState:
    """按实体与数据角色隔离的严格过去状态；先读后写。"""

    def __init__(self, entity_count: int, slots: int, width: int) -> None:
        self.slots = slots
        self.mean = torch.zeros(entity_count, width, dtype=torch.float32)
        self.queue = torch.zeros(entity_count, slots - 1, width, dtype=torch.float32)
        self.count = torch.zeros(entity_count, dtype=torch.int64)
        self.filled = torch.zeros(entity_count, dtype=torch.int64)

    def write(self, rows: torch.Tensor, summary: torch.Tensor) -> None:
        """预测与告警完成后才调用。Welford 在线均值，队列取最近 R-1 条。"""
        self.count[rows] += 1
        delta = summary - self.mean[rows]
        self.mean[rows] += delta / self.count[rows].unsqueeze(1).float()
        self.queue[rows] = torch.cat([self.queue[rows][:, 1:], summary.unsqueeze(1)], dim=1)
        self.filled[rows] = torch.clamp(self.filled[rows] + 1, max=self.slots - 1)
```

- [ ] **步骤 2：实现交叉注意力与标量门**

```python
class CausalEntityMemoryAttention(nn.Module):
    """当前 [CLS] 单向查询严格过去记忆；无历史时返回零上下文，不做全掩码 softmax。"""

    def forward(self, cls_hidden, memory, memory_valid):
        has_history = memory_valid.any(dim=1)
        context = torch.zeros_like(cls_hidden)
        if bool(has_history.any()):
            keys = self.norm_memory(memory[has_history]) + self.role_embedding
            query = self.norm_query(cls_hidden[has_history])
            scores = torch.einsum("bhd,bshd->bhs", self._heads(query, self.w_q),
                                  self._heads_slots(keys, self.w_k)) / math.sqrt(self.head_dim)
            scores = scores.masked_fill(~memory_valid[has_history].unsqueeze(1), float("-inf"))
            scores = scores - scores.max(dim=-1, keepdim=True).values  # 先减行最大值
            weights = scores.softmax(dim=-1)
            values = self._heads_slots(keys, self.w_v)
            merged = torch.einsum("bhs,bshd->bhd", weights, values).flatten(1)
            context[has_history] = self.w_out(merged)
        gate = torch.sigmoid(self.gate(torch.cat([self.norm_query(cls_hidden), context], dim=-1)))
        return cls_hidden + gate * context
```
