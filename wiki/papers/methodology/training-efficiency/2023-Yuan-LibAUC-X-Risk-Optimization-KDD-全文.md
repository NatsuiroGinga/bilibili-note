```python
def GCLoss(**kwargs)
    """Defines dynamic mini-batch loss for GCL."""
    # logits: pairwise similarities, labels: pairwise one
    -hot labels, B: batch size
    neg_logits = exp(logits/tau) * (1-labels)
    u = (1-gamma) * u[index] \
    + gamma * sum(neg_logits, dim=1)/(2(B-1))
    p = (neg_logits/u).detach()
    sum_neg_logits = sum(p * logits, dim=1)/(2(B-1))
    normalized_logits = logits - sum_neg_logits
    loss = -sum(labels * normalized_logits, dim=1)
    return loss.mean()
```
